/* 1. View All Records */

SELECT *
FROM amazon_sales;


/* 2. Calculate Total Sales */

SELECT
    SUM(sales) AS Total_Sales
FROM amazon_sales;


/* 3. Calculate Total Profit */

SELECT
    SUM(profit) AS Total_Profit
FROM amazon_sales;


/* 4. Count Total Orders */

SELECT
    COUNT(order_id) AS Total_Orders
FROM amazon_sales;


/* 5. Sales by Region */

SELECT
    region,
    SUM(sales) AS Total_Sales
FROM amazon_sales
GROUP BY region
ORDER BY Total_Sales DESC;


/* 6. Profit by Region */

SELECT
    region,
    SUM(profit) AS Total_Profit
FROM amazon_sales
GROUP BY region
ORDER BY Total_Profit DESC;


/* 7. Sales by Category */

SELECT
    category,
    SUM(sales) AS Total_Sales
FROM amazon_sales
GROUP BY category
ORDER BY Total_Sales DESC;


/* 8. Profit by Category */

SELECT
    category,
    SUM(profit) AS Total_Profit
FROM amazon_sales
GROUP BY category
ORDER BY Total_Profit DESC;


/* 9. Top 10 Sub-Categories by Sales */

SELECT
    sub_category,
    SUM(sales) AS Total_Sales
FROM amazon_sales
GROUP BY sub_category
ORDER BY Total_Sales DESC
LIMIT 10;


/* 10. Monthly Sales Trend */

SELECT
    YEAR(order_date) AS Year,
    MONTH(order_date) AS Month,
    SUM(sales) AS Total_Sales
FROM amazon_sales
GROUP BY YEAR(order_date), MONTH(order_date)
ORDER BY Year, Month;


/* 11. Payment Method Distribution */

SELECT
    payment_method,
    COUNT(order_id) AS Total_Orders
FROM amazon_sales
GROUP BY payment_method
ORDER BY Total_Orders DESC;


/* 12. Customer Segment Performance */

SELECT
    segment,
    SUM(sales) AS Total_Sales,
    SUM(profit) AS Total_Profit
FROM amazon_sales
GROUP BY segment
ORDER BY Total_Sales DESC;


/* 13. Average Customer Rating */

SELECT
    ROUND(AVG(customer_rating),2) AS Average_Customer_Rating
FROM amazon_sales;


/* 14. Returned Orders */

SELECT
    returned,
    COUNT(order_id) AS Total_Orders
FROM amazon_sales
GROUP BY returned;


/* 15. Profit Margin by Category */

SELECT
    category,
    SUM(sales) AS Total_Sales,
    SUM(profit) AS Total_Profit,
    ROUND((SUM(profit) / SUM(sales)) * 100, 2) AS Profit_Margin_Percentage
FROM amazon_sales
GROUP BY category
ORDER BY Profit_Margin_Percentage DESC;
